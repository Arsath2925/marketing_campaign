df_clean = df.copy()

assert (df_clean["Conversions"] > df_clean["Clicks"]).sum() == 0
assert (df_clean["SpendINR"] < 0).sum() == 0

stage_04_data_manipulation.py
🐍 Python Implementation Code
import numpy as np

# 4. Feature Engineering & Derived Metrics
df_clean["CTR_%"] = (df_clean["Clicks"] / df_clean["Impressions"]) * 100
df_clean["ConversionRate_%"] = (df_clean["Conversions"] / df_clean["Clicks"]) * 100
df_clean["CPC_INR"] = df_clean["SpendINR"] / df_clean["Clicks"]
df_clean["CPA_INR"] = df_clean["SpendINR"] / df_clean["Conversions"]

# Response segmentation by Median CVR
median_cvr = df_clean["ConversionRate_%"].median()
df_clean["ResponseStatus"] = np.where(
    df_clean["ConversionRate_%"] >= median_cvr, "High Responder", "Low Responder")

df_clean["Quarter"] = "Q" + df_clean["CampaignDate"].dt.quarter.astype(str)
top5 = df_clean.sort_values(by="Conversions", ascending=False).head(5)
print(top5[["CampaignID","Channel","AudienceSegment","Conversions","CPA_INR"]])

1
Asymmetry in Output vs. Uniformity in Budgeting

Spend skewness is near-zero (-0.01), proving budgets were allocated relatively uniformly across tests. However, Conversions are heavily right-skewed (+1.56), showing that marketing outputs follow a power-law distribution where a small group of campaigns drive outsized impact.

2
Identification of 12 Statistical Outliers via IQR

Applying the 1.5 × IQR threshold isolated 12 upper-bound outliers exceeding 1,489 conversions. Rather than errors, these represent highly successful, viral campaigns that provide clear templates for future budget scaling.

stage_05_advanced_analysis.py
🐍 Python Implementation Code
# 5. Skewness, Outliers & Channel Breakdown
metrics = ["Impressions", "Clicks", "Conversions", "SpendINR"]
print("Metric Skewness:\n", df_clean[metrics].skew().round(2))

# IQR Outlier Detection
q25, q75 = df_clean["Conversions"].quantile(0.25), df_clean["Conversions"].quantile(0.75)
iqr = q75 - q25
outliers = df_clean[df_clean["Conversions"] > (q75 + 1.5 * iqr)]
print("Outlier Campaigns Count:", len(outliers))

# Grouping by Channel
ch_summary = df_clean.groupby("Channel", observed=False).agg(
    Runs=("CampaignID", "count"),
    Spend=("SpendINR", "sum"),
    Conversions=("Conversions", "sum"),
    Avg_CTR=("CTR_%", "mean"),
    Avg_CVR=("ConversionRate_%", "mean"))
print(ch_summary)
